# Python ka lightweight official image use kar rahe hain
FROM python:3.10-slim

# Working directory set kar rahe hain
WORKDIR /app

# System dependencies install karna agar zaroori ho
RUN apt-get update && apt-get install -y \
    build-essential \
    curl \
    software-properties-common \
    git \
    && rm -rf /var/lib/apt/lists/*

# Requirements file copy karo aur dependencies install karo
COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt

# Baaki ka poora project code copy karo
COPY . .

# Streamlit ya jo bhi main script ho uska port expose karo
EXPOSE 8501

# Health check taaki app status pata rahe
HEALTHCHECK CMD curl --fail http://localhost:8501/_stcore/health || exit 1

# Command to run the application (agar app.py ya streamlit app hai)
# Agar aapka main file app.py hai toh ye use hoga:
CMD ["streamlit", "run", "app.py", "--server.port=8501", "--server.address=0.0.0.0"]
